import React, { useState } from 'react';
import {
  SafeAreaView,
  View,
  Text,
  TextInput,
  Pressable,
  StyleSheet,
  ScrollView,
  Alert,
} from 'react-native';

export default function App() {
  const [servicio, setServicio] = useState('Tienda');
  const [direccion, setDireccion] = useState('');
  const [destino, setDestino] = useState('');
  const [km, setKm] = useState('1');

  const distancia = Math.max(0, Number(km) || 0);
  const precio = 30 + distancia * 10;

  return (
    <SafeAreaView style={styles.fondo}>
      <ScrollView contentContainerStyle={styles.contenido}>
        <Text style={styles.logo}>🚲</Text>
        <Text style={styles.titulo}>Mandados Santiago</Text>
        <Text style={styles.subtitulo}>
          Tus mandados, hasta tu puerta
        </Text>

        <Text style={styles.etiqueta}>¿Qué necesitas?</Text>
        <View style={styles.servicios}>
          {['Tienda', 'Comida', 'Farmacia', 'Paquetes'].map((s) => (
            <Pressable
              key={s}
              onPress={() => setServicio(s)}
              style={[
                styles.opcion,
                servicio === s && styles.seleccionado,
              ]}
            >
              <Text style={servicio === s ? styles.blanco : styles.texto}>
                {s}
              </Text>
            </Pressable>
          ))}
        </View>

        <Text style={styles.etiqueta}>Lugar de recogida</Text>
        <TextInput
          style={styles.campo}
          placeholder="Tienda o dirección"
          value={direccion}
          onChangeText={setDireccion}
        />

        <Text style={styles.etiqueta}>¿A dónde lo llevamos?</Text>
        <TextInput
          style={styles.campo}
          placeholder="Tu dirección de entrega"
          value={destino}
          onChangeText={setDestino}
        />

        <Text style={styles.etiqueta}>Distancia estimada en km</Text>
        <TextInput
          style={styles.campo}
          placeholder="Ej. 3"
          keyboardType="decimal-pad"
          value={km}
          onChangeText={setKm}
        />

        <View style={styles.precio}>
          <Text style={styles.precioTexto}>Precio estimado</Text>
          <Text style={styles.cantidad}>
            ${Number.isFinite(precio) ? precio.toFixed(2) : '30.00'} MXN
          </Text>
          <Text style={styles.nota}>
            $30 base + $10 por kilómetro.
            Precio por confirmar.
          </Text>
        </View>

        <Pressable
          style={styles.boton}
          onPress={() => {
            if (!direccion.trim() || !destino.trim()) {
              Alert.alert(
                'Faltan datos',
                'Escribe el lugar de recogida y la dirección de entrega.'
              );
              return;
            }
            Alert.alert(
              'Demostración',
              'Tu formulario funciona. Todavía no se envía un pedido real.'
            );
          }}
        >
          <Text style={styles.botonTexto}>Solicitar mandado</Text>
        </Pressable>
        <Text style={styles.pie}>
          Santiago Papasquiaro, Durango
        </Text>
      </ScrollView>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  fondo: { flex: 1, backgroundColor: '#f4f7f2' },
  contenido: { padding: 22, paddingTop: 32 },
  logo: { fontSize: 46, textAlign: 'center' },
  titulo: {
    fontSize: 27, fontWeight: 'bold',
    textAlign: 'center', color: '#174c32',
  },
  subtitulo: {
    textAlign: 'center', color: '#647067',
    marginTop: 7, marginBottom: 26,
  },
  etiqueta: {
    fontWeight: '600', color: '#25372b',
    marginBottom: 8, marginTop: 15,
  },
  servicios: {
    flexDirection: 'row', flexWrap: 'wrap', gap: 8,
  },
  opcion: {
    backgroundColor: '#fff', padding: 12,
    borderRadius: 10, borderWidth: 1, borderColor: '#d8e1d8',
  },
  seleccionado: { backgroundColor: '#247747', borderColor: '#247747' },
  texto: { color: '#25372b' },
  blanco: { color: '#fff', fontWeight: 'bold' },
  campo: {
    backgroundColor: '#fff', borderWidth: 1,
    borderColor: '#d8e1d8', borderRadius: 5,
    padding: 14, fontSize: 16,
  },
  precio: {
    marginTop: 22, padding: 18, borderRadius: 14,
    backgroundColor: '#e2efdf',
  },
  precioTexto: { color: '#31513b' },
  cantidad: {
    fontSize: 29, fontWeight: 'bold',
    color: '#174c32', marginTop: 5,
  },
  nota: { color: '#526455', marginTop: 6 },
  boton: {
    backgroundColor: '#247747', padding: 16,
    borderRadius: 12, alignItems: 'center', marginTop: 20,
  },
  botonTexto: { color: '#fff', fontWeight: 'bold', fontSize: 16 },
  pie: { textAlign: 'center', color: '#68756a', marginTop: 22 },
});
